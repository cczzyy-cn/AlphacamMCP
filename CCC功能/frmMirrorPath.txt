' ==============================================================================
' frmMirrorPath -- 路径自身镜像对话框 (UserForm)
' ==============================================================================
' 使用方法：
'   1. 在 VBA 编辑器中插入 UserForm（名称设为 frmMirrorPath）
'   2. 从工具箱添加以下控件（名称必须完全一致）：
'      +------------------+---------------+-------------------------------+
'      | 控件名            | 类型          | 说明                          |
'      +------------------+---------------+-------------------------------+
'      | optMirrorX       | OptionButton  | 绕X轴镜像（上下翻转，默认）  |
'      | optMirrorY       | OptionButton  | 绕Y轴镜像（左右翻转）        |
'      | cmdOK            | CommandButton | 确定按钮                      |
'      | cmdCancel        | CommandButton | 取消按钮                      |
'      +------------------+---------------+-------------------------------+
'   3. 双击窗体 -> "查看代码" -> 粘贴本文件全部内容
' ==============================================================================
Option Explicit

' --------------------------------------------------------------------------
' 模块级常量
' --------------------------------------------------------------------------
Private Const REG_APP As String = "CCC_MirrorPath"
Private Const REG_SECTION As String = "MirrorPath"

' --------------------------------------------------------------------------
' Windows API 声明 - 为对话框添加最小化按钮
' --------------------------------------------------------------------------
Private Declare PtrSafe Function FindWindow Lib "user32" Alias "FindWindowA" _
    (ByVal lpClassName As String, ByVal lpWindowName As String) As Long
Private Declare PtrSafe Function GetWindowLong Lib "user32" Alias "GetWindowLongA" _
    (ByVal hWnd As Long, ByVal nIndex As Long) As Long
Private Declare PtrSafe Function SetWindowLong Lib "user32" Alias "SetWindowLongA" _
    (ByVal hWnd As Long, ByVal nIndex As Long, ByVal dwNewLong As Long) As Long
Private Declare PtrSafe Function SetWindowPos Lib "user32" _
    (ByVal hWnd As Long, ByVal hWndInsertAfter As Long, _
     ByVal X As Long, ByVal Y As Long, ByVal cx As Long, ByVal cy As Long, _
     ByVal wFlags As Long) As Long

Private Const GWL_STYLE As Long = (-16)
Private Const WS_MINIMIZEBOX As Long = &H20000
Private Const SWP_NOSIZE As Long = &H1
Private Const SWP_NOMOVE As Long = &H2
Private Const SWP_NOZORDER As Long = &H4
Private Const SWP_FRAMECHANGED As Long = &H20


' ==============================================================================
' UserForm 事件
' ==============================================================================

Private Sub UserForm_Initialize()
    Me.Caption = "路径自身镜像"
    
    ' 显示已选中路径数
    Dim tpCount As Long: tpCount = 0
    Dim pc As Collection: Set pc = modMirrorPath.GetSelectedPaths
    If Not (pc Is Nothing) Then tpCount = pc.Count
    
    Dim info As String
    If tpCount > 0 Then
        info = "已选中 " & tpCount & " 条刀具路径"
    Else
        info = "未选中路径"
    End If
    
    ' 添加标签显示路径数（用窗体标题替代--VBA UserForm 不支持运行时添加控件）
    ' 改用 Caption 显示路径数
    Me.Caption = "路径自身镜像 - " & info
    
    ' 读取上次选择的镜像轴
    Dim lastAxis As String
    lastAxis = GetSetting(REG_APP, REG_SECTION, "MirrorAxis", "X")
    If lastAxis = "X" Then
        optMirrorX.Value = True
    Else
        optMirrorY.Value = True
    End If
End Sub

Private Sub UserForm_Activate()
    AddMinButton Me
End Sub


' ==============================================================================
' 确定 - 执行路径镜像
' ==============================================================================
Private Sub cmdOK_Click()
    Me.Hide
    DoEvents
    
    ' 调用镜像逻辑
    DoMirrorPath optMirrorX.Value
    
    ' 保存设置到注册表
    If optMirrorX.Value Then
        SaveSetting REG_APP, REG_SECTION, "MirrorAxis", "X"
    Else
        SaveSetting REG_APP, REG_SECTION, "MirrorAxis", "Y"
    End If
    
    Unload Me
End Sub


' ==============================================================================
' 取消
' ==============================================================================
Private Sub cmdCancel_Click()
    Unload Me
End Sub


' ==============================================================================
' AddMinButton - 为窗体添加最小化按钮
' ==============================================================================
Private Sub AddMinButton(frm As Object)
    Dim hWnd As Long
    hWnd = FindWindow("ThunderDFrame", frm.Caption)
    If hWnd = 0 Then hWnd = FindWindow("ThunderRT5Form", frm.Caption)
    If hWnd = 0 Then Exit Sub
    Dim style As Long
    style = GetWindowLong(hWnd, GWL_STYLE)
    style = style Or WS_MINIMIZEBOX
    SetWindowLong hWnd, GWL_STYLE, style
    SetWindowPos hWnd, 0, 0, 0, 0, 0, _
        SWP_NOMOVE Or SWP_NOSIZE Or SWP_NOZORDER OR SWP_FRAMECHANGED
End Sub
